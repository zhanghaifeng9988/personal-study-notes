-- 查询表空间使用率
col TABLESPACE_NAME for a20
col name for a10
set line 200 pages 999
select TABLESPACE_NAME,round(USED_SPACE*8/1024, 2) AS "USED-M",
round(TABLESPACE_SIZE*8/1024, 2) AS "SIZE-M",round((TABLESPACE_SIZE-USED_SPACE)*8/1024, 2) "FREE-M",
to_char(USED_PERCENT,'00.00') as "USED_PERCENT"  
from dba_tablespace_usage_metrics order by used_percent desc  ;


--磁盘组使用情况带百分号 
col USED_PERCENT for a10
select state, name, type, ROUND(total_mb/1024,2) AS "TOTAL_G", round(free_mb/1024 ,2) AS "FREE_G",
concat(round((1-(round(free_mb/1024 ,2))/(ROUND(total_mb/1024,2)))*100,2) , '%') AS "USED_PERCENT"
from v$asm_diskgroup_stat;

--磁盘组使用情况不带百分号
select state, name, type, ROUND(total_mb/1024,2) AS "TOTAL_G", round(free_mb/1024 ,2) AS "FREE_G",
round((round(free_mb/1024 ,2))/(ROUND(total_mb/1024,2))*100,2)  AS "USED_PERCENT"
from v$asm_diskgroup_stat;
