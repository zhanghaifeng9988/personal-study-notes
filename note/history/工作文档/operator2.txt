--查看当前有哪些用户正在使用数据库
select osuser,a.username,cpu_time/executions/1000000||'s',b.sql_text,machine
from v$session a,v$sqlarea b 
where a.sql_address=b.address
order by cpu_time/executions desc;

--当前的session连接数  单实例
select count(*) from v$session;
 
 rac场景 多实例
select count(*) from gv$session where wait_class#<>6 and type='USER';

session:指定了一个实例允许的会话数，即同时登录到数据库的并发用户数；
process：指定了一个实例可以同时运行的进程数
session的值通过process的值计算得到，11g session=(1.5*process)+22
session 通常要大于 process
