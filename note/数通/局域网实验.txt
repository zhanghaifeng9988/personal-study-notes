   此时两个网段到192.168.100.1也都是通的
  
  备注：vlan默认不同，可以通过配置静态路由、VLAN间路由或在三层交换机上设置VLANIF接口等方式来实现。
  vlan间路由的一种方式：（单臂路由）在交换机和路由器之间仅使用一条物理链路连接。在交换机上，
  把连接到路由器的端口配置成Trunk类型的端口，并允许相关VLAN的帧通过。
  在路由器上需要创建子接口，逻辑上把连接路由器的物理链路分成了多条，
  每个子接口代表了一条归属于某个VLAN的逻辑链路。
  
  如果需要VLAN内的所有用户不能通过VLANIF接口与其他VLAN内用户通信，但VLAN内用户还可以互相通信，
  可以在VLANIF接口视图下执行命令shutdown来禁止三层流量，但不影响二层流量。
  
  
--局域网路由配置  OSPF路由模式
   --路由器，与核心交换机连接接口配置
   interface GigabitEthernet 0/0/1
   ip address 10.1.100.2 24

   
   
   --core核心交换机ospf配置
   interface GigabitEthernet 0/0/1
   port link-type access
   port default vlan 100
   
   router id 192.168.100.1
   ospf 1
   area 0
   network 192.168.10.0  0.0.0.255
   network 192.168.20.0  0.0.0.255
   network 192.168.30.0  0.0.0.255
   network 192.168.40.0  0.0.0.255
   network 192.168.100.1  0.0.0.0
   
   --路由器ospf配置
   router id 192.168.100.2
   ospf 1
   area 0
   network 192.168.100.2 0.0.0.0 
   
   --查看邻居
   display ospf peer
   --查看ospf的路由
   display ospf routing  或者dis ip routing-table  protocol  ospf
   

--到公网配置路由
  --路由器  配置静态路由到ISP
    interface GigabitEthernet 0/0/0
    ip address 1.1.1.1 24
	
    ip route-static 0.0.0.0  0  1.1.1.2
	
	--用于形成内网到路由器的静态路由
	ospf 1  
   default-route-advertise
	
--nat（easyip模式）
   --路由器
     acl 2000
   rule permit source 192.168.10.0  0.0.0.255  --允许10网段访问互联网
   rule deny  source 192.168.20.0   0.0.0.255  --组织20网段访问互联网
   rule deny  source 192.168.30.1   0.0.0.0    --阻止特定ip访问互联网
   interface GigabitEthernet 0/0/0
   nat outbound 2000
   
--nat地址映射，让外网可以访问局域网内部服务器
   --ISP路由器配置
   interface GigabitEthernet 0/0/1
   ip address 200.1.1.2 24
   --给测试电脑配置ip 200.1.1.1  24
   --给测试服务器配置ip 192.168.30.1
   --给测试服务器配置http服务

   --路由器配置（可以使外网用户访问局域网服务器）
   进入公网口
   interface GigabitEthernet 0/0/0
   nat server protocol tcp global 1.1.1.10 80 inside 192.168.30.1 80
   

  --使内网用户，通过公网ip访问内部服务器
    路由器配置
  acl 3000
  rule permit ip source 192.168.0.0 0.0.255.255 destination  1.1.1.10 0
  interface GigabitEthernet 0/0/1
  nat outbound 3000
  nat server protocol tcp global 1.1.1.10 80 inside 192.168.30.1 80
  
  
  注意：当然内网用户直接用内网地址，也可以访问服务器
