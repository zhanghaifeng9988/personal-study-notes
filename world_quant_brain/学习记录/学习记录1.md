


# 显示当前在哪个环境（带*的是当前环境）
conda info --envs 
conda env list

# 切换python环境
 conda activate base

# 查看当前环境安装了哪些包
conda list

# 删除环境
 conda remove -n myenv --all

# 创建新的环境
conda create -n quant_study python=3.11 numpy pandas matplotlib jupyter -y


#  删除后,创建新环境后，清理包缓存，释放空间
conda clean --all
